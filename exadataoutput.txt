SQL> WITH cfg AS (
  SELECT /*+ MATERIALIZE */ cellname, conftype, XMLTYPE(confval) x
  3    FROM   v$cell_config
  WHERE  conftype IN ('CELL','PHYSICALDISKS','CELLDISKS','GRIDDISKS','LUNS')
),
  6  cell AS (
  7    SELECT c.cellname, t.cell_host
  8    FROM   cfg c, XMLTABLE('/cli-output/cell' PASSING c.x
           COLUMNS cell_host VARCHAR2(64) PATH 'name') t
 10    WHERE  c.conftype = 'CELL'
),
 12  gd AS (
 13    SELECT c.cellname, t.*
 14    FROM   cfg c, XMLTABLE('/cli-output/griddisk' PASSING c.x
           COLUMNS gd_name   VARCHAR2(64) PATH 'name',
                   celldisk  VARCHAR2(64) PATH 'cellDisk',
                   asm_name  VARCHAR2(64) PATH 'asmDiskName',
                   gd_size   VARCHAR2(30) PATH 'size',
                   gd_status VARCHAR2(30) PATH 'status') t
 20    WHERE  c.conftype = 'GRIDDISKS'
),
 22  cd AS (
 23    SELECT c.cellname, t.*
 24    FROM   cfg c, XMLTABLE('/cli-output/celldisk' PASSING c.x
           COLUMNS cd_name   VARCHAR2(64) PATH 'name',
                   cd_type   VARCHAR2(30) PATH 'diskType',
                   lun       VARCHAR2(64) PATH 'lun',
                   physdisk  VARCHAR2(64) PATH 'physicalDisk',
                   devpart   VARCHAR2(64) PATH 'devicePartition') t
 30    WHERE  c.conftype = 'CELLDISKS'
),
 32  lun AS (
 33    SELECT c.cellname, t.*
 34    FROM   cfg c, XMLTABLE('/cli-output/lun' PASSING c.x
           COLUMNS lun_id    VARCHAR2(64) PATH 'id',
                   celldisk  VARCHAR2(64) PATH 'cellDisk',
                   lun_dev   VARCHAR2(64) PATH 'deviceName',
                   wcache    VARCHAR2(40) PATH 'lunWriteCacheMode') t
 39    WHERE  c.conftype = 'LUNS'
),
 41  pd AS (
 42    SELECT c.cellname, t.*
 43    FROM   cfg c, XMLTABLE('/cli-output/physicaldisk' PASSING c.x
           COLUMNS pd_id     VARCHAR2(64) PATH 'id',
                   pd_name   VARCHAR2(64) PATH 'name',
                   pd_type   VARCHAR2(30) PATH 'diskType',
                   slot      VARCHAR2(30) PATH 'slotNumber',
                   pd_status VARCHAR2(60) PATH 'status') t
 49    WHERE  c.conftype = 'PHYSICALDISKS'
)
 51  SELECT adg.name                           asm_diskgroup,
 52         ad.name                            asm_disk,
 53         ad.mode_status                     asm_mode,
 54         gd.gd_name                         griddisk,
 55         gd.gd_size,
 56         cd.cd_name                         celldisk,
 57         cd.cd_type,
       NVL(cell.cell_host, gd.cellname)   cell_host,
 59         lun.lun_dev,
 60         pd.pd_name                         physdisk,
 61         pd.slot,
       SUBSTR(pd.pd_status,1,14)          pd_status
 63  FROM   gd
 64  LEFT JOIN cell ON cell.cellname = gd.cellname
 65  LEFT JOIN v$asm_disk_stat ad
 66         ON ad.path = 'o/' || gd.cellname || '/' || gd.gd_name
 67  LEFT JOIN v$asm_diskgroup_stat adg ON adg.group_number = ad.group_number
 68  LEFT JOIN cd  ON cd.cellname  = gd.cellname AND cd.cd_name  = gd.celldisk
LEFT JOIN lun ON lun.cellname = cd.cellname AND (lun.celldisk = cd.cd_name OR lun.lun_id = cd.lun)
LEFT JOIN pd  ON pd.cellname  = cd.cellname AND (pd.pd_id = cd.physdisk OR pd.pd_name = cd.physdisk)
 71  ORDER  BY asm_diskgroup, cell_host, celldisk, griddisk;
